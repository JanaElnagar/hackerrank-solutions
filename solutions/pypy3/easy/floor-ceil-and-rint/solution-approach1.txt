# Enter your code here. Read input from STDIN. Print output to STDOUT
import numpy as np
np.set_printoptions(legacy='1.13')

input_str = input()
a = np.array(list(map(float,input_str.split())))


print(np.floor(a))
print(np.ceil(a))
print(np.rint(a))
