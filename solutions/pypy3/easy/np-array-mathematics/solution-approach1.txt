# Enter your code here. Read input from STDIN. Print output to STDOUT
import numpy as np
np.set_printoptions(legacy='1.13')

input1 = input()

[N,M] = map(int, input1.split())

list_a = []
for i in range(N):
    line = input()
    list_a.append(list(map(int, line.split())))
    
a = np.array(list_a)
    
list_b = []
for i in range(N):
    line = input()
    list_b.append(list(map(int, line.split())))
    
b = np.array(list_b)
    
    
    
print(a+b)
print(a-b)
print(a*b)
print(a//b)
print(a%b)
print(a**b)
